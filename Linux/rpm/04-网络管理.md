## 修改网卡

``` bash
# 配置eth1网络接口:
[root@rockylinux ~]# nmcli connection show
NAME                UUID                                  TYPE      DEVICE 
Wired connection 1  be7f1f19-9a56-3764-af07-04ed70fee9fc  ethernet  -- 

# 生成配置:
[root@rockylinux ~]# nmcli connection add con-name eth1 type ethernet ifname eth1
Connection 'eth1' (3d159030-30a5-4cc1-aa19-f2ebf106262e) successfully added.
	con-name            名称自定义
	ifname              名称一定是内核所识别设备名

# 使用nmcli修改eth1网卡配置文件
[root@rockylinux ~]# nmcli connection modify eth1 ipv4.addresses 172.16.1.1/16 ipv4.gateway 172.16.1.254 ipv4.method manual
	modify eth1         con-name设备连接名称
	ipv4.addresses      配置IP地址
	ipv4.method         manual 静态IP

# 激活接口
[root@rockylinux ~]# nmcli connection up eth1
Connection successfully activated (D-Bus active path: /org/freedesktop/NetworkManager/ActiveConnection/26)
```

#### ethtool工具的使用
``` bash
ethtool
	-i    查看网卡驱动信息    
	-t    网卡自检
	-S    统计网卡是否丢包
    -s    将网卡 eth0 强制设置为 100Mbps(百兆),全双工(Full Duplex)并且关闭自动协商(Autoneg)
		  ethtool -s eth0 speed 100 duplex full autoneg off 
          设置网卡唤醒
		  ethtool eth0 | grep Wake-on
	-P    查看mac地址
	-p    查看网卡对应的物理设备接口是哪个
```

#### lsof工具的使用
``` bash
# 追踪命令打开了哪些函数
[root@rockylinux ~]# strace tail

# 查看文件被哪些命令使用
[root@rockylinux ~]# lsof /etc/passwd
[root@rockylinux ~]# lsof | grep /etc/passwd

# 查看PID进程号
[root@rockylinux ~]# lsof -p 14557

# 查看端口号
[root@rockylinux ~]# lsof -i :22
[root@rockylinux ~]# lsof -i :80

# 查看用户打开的文件
[root@rockylinux ~]# lsof -u root
[root@rockylinux ~]# lsof -u nginx

# 删除文件后, 磁盘不被释放
[root@rockylinux ~]# lsof | grep deleted
[root@rockylinux ~]# ll /proc/14385/fd
[root@rockylinux ~]# echo > /proc/14385/fd/3
```
