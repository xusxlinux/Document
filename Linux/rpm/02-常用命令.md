``` shell
history                         记录历史命令, 忽略敏感命令
timedatectl                     查看和修改时区 timedatectl set-ntp true 启动系统默认的网络时间同步服务
hostnamectl                     设置主机名
alias                           命令别名
date                            时间戳: echo `date +%y-%m-%d-%H:%M:%S`
passwd                          修改当前用户密码: 
file                            查看文件类型
cat                             查看文本内容
more/less                       分页查看      
head/tail                       查看前10行
wc                              统计文件行数, 字符数
ls/ll                           显示文件列表
whatis                          查看命令的用法
man                             帮助文档的使用 
mandb                           更新man索引库

cd                              返回当前用户的家目录
cp                              拷贝 -r递归
mv                              重命名 移动剪切
scp                             openssh加密 跨主机的方式拷贝 -r拷贝目录
rsync                           快速拷贝  增量拷贝/备份 与定时任务设置同步策略 -P(显示拷贝过程)
touch                           创建文本
mkdir                           创建目录
rm                              删除文件 rm -rf
ln                              硬链接用于数据备份,目录是不能创建硬链接的

chown                           chown -R xusx:xusx file
chgrp                           chgrp xusx file
chmod                           修改权限 r=4 w=2 x=1 
setuid                          特殊权限,仅针对文件设置, 该文件必须是程序,且有x权限
setgit                          针对目录, 必须对目录有wx权限, 内部创新的文件将和父目录有相同的组

w                               查看用户登录会话
ps                              aux
pstree                          以树状图的形式, 直观地显示系统中所有正在运行的进程之间的父子关系
pgrep -l -u xusx                查看xusx用户所有进程
pidof                           查看服务的pid  pidof systemd
mpstat -P all 1                 监控所有CPU 1秒钟输出
jobs fg[1]/bg[1]                后台运行程序

kill                            常用信号是 1重启 9终止 18激活 19暂停 kill -1 $(pidof systemd)
pkill                           管理进程组
killall                         管理同名进程

pkill -SIGKILL -u xusx          向xusx用户所有进程发送kill信号
pkill -SIGKILL -t tty3          介绍终端会话
```
